# ==============================================================================
# PJSIP Cross-Platform Configuration
# ==============================================================================

# Define base path in the local library repository
set(PJSIP_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../QtLibraries/pjsip" CACHE PATH "Path to PJSIP root directory")

# Add include directory for all platforms
target_include_directories(${PROJECT_NAME} PRIVATE
    ${PJSIP_ROOT}/include
)

# ------------------------------------------------------------------------------
# 1. Windows (MinGW x86_64)
# ------------------------------------------------------------------------------
if(WIN32)
    message(STATUS "[PJSIP] Configuring for Windows MinGW x86_64...")

    target_link_directories(${PROJECT_NAME} PRIVATE
        ${PJSIP_ROOT}/lib/windows-x86_64
    )

    # Name suffix of binaries generated by MinGW
    set(PJSIP_SUFFIX "-x86_64-w64-mingw32")

    target_link_libraries(${PROJECT_NAME} PRIVATE
        # PJSUA2 (C++ Object-Oriented API)
        pjsua2${PJSIP_SUFFIX}

        # Core C API and SIP protocol stack
        pjsua${PJSIP_SUFFIX}
        pjsip-ua${PJSIP_SUFFIX}
        pjsip-simple${PJSIP_SUFFIX}
        pjsip${PJSIP_SUFFIX}

        # Media stack (Audio and Video processing)
        pjmedia-codec${PJSIP_SUFFIX}
        pjmedia-videodev${PJSIP_SUFFIX}
        pjmedia-audiodev${PJSIP_SUFFIX}
        pjmedia${PJSIP_SUFFIX}

        # NAT Traversal and base utility libraries
        pjnath${PJSIP_SUFFIX}
        pjlib-util${PJSIP_SUFFIX}
        pj${PJSIP_SUFFIX}

        # Codecs and bundled third-party libraries
        webrtc${PJSIP_SUFFIX}
        gsmcodec${PJSIP_SUFFIX}
        resample${PJSIP_SUFFIX}
        srtp${PJSIP_SUFFIX}
        speex${PJSIP_SUFFIX}
        ilbccodec${PJSIP_SUFFIX}
        g7221codec${PJSIP_SUFFIX}

        # Windows system libraries
        ws2_32
        wsock32
        ole32
        dsound
        winmm
        iphlpapi
        avrt
        stdc++
    )

# ------------------------------------------------------------------------------
# 2. Android (NDK)
# ------------------------------------------------------------------------------
elseif(ANDROID)
    message(STATUS "[PJSIP] Configuring for Android (${ANDROID_ABI})...")

    # Automatically route based on the selected architecture in Qt Creator (arm64-v8a or armeabi-v7a)
    target_link_directories(${PROJECT_NAME} PRIVATE
        ${PJSIP_ROOT}/lib/android-${ANDROID_ABI}
    )

    # Library name suffix for Android based on target build triple
    set(PJSIP_SUFFIX "-${CMAKE_ANDROID_ARCH_TRIPLE}")

    target_link_libraries(${PROJECT_NAME} PRIVATE
        # PJSIP libraries
        pjsua2${PJSIP_SUFFIX}
        pjsua${PJSIP_SUFFIX}
        pjsip-ua${PJSIP_SUFFIX}
        pjsip-simple${PJSIP_SUFFIX}
        pjsip${PJSIP_SUFFIX}
        pjmedia-codec${PJSIP_SUFFIX}
        pjmedia-videodev${PJSIP_SUFFIX}
        pjmedia-audiodev${PJSIP_SUFFIX}
        pjmedia${PJSIP_SUFFIX}
        pjnath${PJSIP_SUFFIX}
        pjlib-util${PJSIP_SUFFIX}
        pj${PJSIP_SUFFIX}

        # Bundled codecs and processing modules
        webrtc${PJSIP_SUFFIX}
        gsmcodec${PJSIP_SUFFIX}
        resample${PJSIP_SUFFIX}
        srtp${PJSIP_SUFFIX}
        speex${PJSIP_SUFFIX}
        ilbccodec${PJSIP_SUFFIX}
        g7221codec${PJSIP_SUFFIX}

        # Android NDK system libraries
        OpenSLES
        log
        android
        m
    )

# ------------------------------------------------------------------------------
# 3. Linux Desktop (If building directly on Debian/Ubuntu)
# ------------------------------------------------------------------------------
elseif(UNIX AND NOT ANDROID)
    message(STATUS "[PJSIP] Configuring for Linux...")

    target_link_directories(${PROJECT_NAME} PRIVATE
        ${PJSIP_ROOT}/lib/linux-x86_64
    )

    target_link_libraries(${PROJECT_NAME} PRIVATE
        pjsua2
        pjsua
        pjsip-ua
        pjsip-simple
        pjsip
        pjmedia-codec
        pjmedia-videodev
        pjmedia-audiodev
        pjmedia
        pjnath
        pjlib-util
        pj
        pthread
        dl
        m
    )
endif()
